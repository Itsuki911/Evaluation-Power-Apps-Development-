Private Sub Form_Load()
    Dim argEmp As String
    Dim argYM As String

    argEmp = ParseOpenArgToken(Me.OpenArgs, 0)
    argYM = ParseOpenArgToken(Me.OpenArgs, 1)

    If Len(argYM) = 7 Then
        Me.txtTargetYM.Value = argYM
    Else
        Me.txtTargetYM.Value = Format$(Date, "yyyy-mm")
    End If

    Me.cboFilterStatus.Value = "All"
    Me.txtFilterWorkDate.Value = ""

    If Len(argEmp) > 0 Then
        Me.cboFilterEmployee.Value = argEmp
    Else
        Me.cboFilterEmployee.Value = Null
    End If

    ApplyAttendanceFilter
End Sub

Public Sub ApplyAttendanceFilter()
    Dim ymText As String
    Dim empId As String
    Dim workDateText As String
    Dim statusVal As String
    Dim errMsg As String
    Dim subFilter As String

    ymText = Trim$(Nz(Me.txtTargetYM.Value, ""))
    empId = Trim$(Nz(Me.cboFilterEmployee.Value, ""))
    workDateText = Trim$(Nz(Me.txtFilterWorkDate.Value, ""))
    statusVal = Trim$(Nz(Me.cboFilterStatus.Value, "All"))

    If Len(ymText) > 0 Then
        If Not ValidateYMFilter(ymText, errMsg) Then
            MsgBox errMsg, vbExclamation, "年月指定エラー"
            Me.txtTargetYM.SetFocus
            Exit Sub
        End If
    End If

    If Not ValidateDateFilter(workDateText, errMsg) Then
        MsgBox errMsg, vbExclamation, "日付指定エラー"
        Me.txtFilterWorkDate.SetFocus
        Exit Sub
    End If

    Me.txtHiddenEmpID.Value = empId
    TempVars.Add "TargetYM", ymText

    subFilter = ""
    If Len(empId) > 0 Then
        subFilter = AppendCond(subFilter, "EmployeeID = '" & SqlEscape(empId) & "'")
    End If
    If Len(workDateText) > 0 Then
        subFilter = AppendCond(subFilter, "WorkDate = '" & SqlEscape(workDateText) & "'")
    End If
    If Len(statusVal) > 0 And statusVal <> "All" Then
        subFilter = AppendCond(subFilter, "Status = '" & SqlEscape(statusVal) & "'")
    End If

    With Me.subDetail.Form
        If Len(subFilter) > 0 Then
            .Filter = subFilter
            .FilterOn = True
        Else
            .Filter = ""
            .FilterOn = False
        End If
        .Requery
    End With

    UpdateMonthlyKpiSummary ymText, empId
    Me.Recalc
End Sub

Private Sub UpdateMonthlyKpiSummary(ByVal ymText As String, ByVal empId As String)
    Dim crit As String
    Dim totWork As Long
    Dim totOT As Long
    Dim totLN As Long
    Dim totLeave As Long

    crit = ""
    If Len(ymText) > 0 Then
        crit = AppendCond(crit, "WorkYM = '" & SqlEscape(ymText) & "'")
    End If
    If Len(empId) > 0 Then
        crit = AppendCond(crit, "EmployeeID = '" & SqlEscape(empId) & "'")
    End If

    totWork = CLng(Nz(DSum("TotalWorkMin", "qryMonthlySummary", crit), 0))
    totOT = CLng(Nz(DSum("TotalOvertimeMin", "qryMonthlySummary", crit), 0))
    totLN = CLng(Nz(DSum("TotalLateNightMin", "qryMonthlySummary", crit), 0))
    totLeave = CLng(Nz(DSum("PaidLeaveDays", "qryMonthlySummary", crit), 0))

    Me.txtSumWorkHHMM.Value = FormatMinutesHHMM(totWork)
    Me.txtSumOvertimeHHMM.Value = FormatMinutesHHMM(totOT)
    Me.txtSumLateNightHHMM.Value = FormatMinutesHHMM(totLN)
    Me.txtSumPaidLeave.Value = totLeave
End Sub

Private Function AppendCond(ByVal baseExpr As String, ByVal addExpr As String) As String
    If Len(baseExpr) = 0 Then
        AppendCond = addExpr
    Else
        AppendCond = baseExpr & " AND " & addExpr
    End If
End Function

Private Sub btnOpenCrosstab_Click()
    Dim ymText As String
    Dim errMsg As String
    ymText = Trim$(Nz(Me.txtTargetYM.Value, ""))
    If Not ValidateYMFilter(ymText, errMsg) Then
        MsgBox errMsg, vbExclamation, "対象年月未指定"
        Exit Sub
    End If
    ' Opens crosstab query qryMonthlyCrosstab which reads [Forms]![frmAttendance]![txtTargetYM]
    DoCmd.OpenQuery "qryMonthlyCrosstab", acViewNormal, acReadOnly
End Sub

Private Sub btnPrintReport_Click()
    Dim whereCond As String
    Dim ymText As String
    Dim empId As String

    whereCond = ""
    ymText = Trim$(Nz(Me.txtTargetYM.Value, ""))
    empId = Trim$(Nz(Me.cboFilterEmployee.Value, ""))

    If Len(ymText) > 0 Then
        whereCond = AppendCond(whereCond, "WorkYM = '" & SqlEscape(ymText) & "'")
    End If
    If Len(empId) > 0 Then
        whereCond = AppendCond(whereCond, "EmployeeID = '" & SqlEscape(empId) & "'")
    End If

    DoCmd.OpenReport "rptAttendanceHistory", acViewPreview, , whereCond
End Sub

Private Sub btnBackToMain_Click()
    If Not CurrentProject.AllForms("frmMain").IsLoaded Then
        DoCmd.OpenForm "frmMain", acNormal
    Else
        Forms("frmMain").RefreshAttendanceViews
        Forms("frmMain").SetFocus
    End If
    DoCmd.Close acForm, Me.Name, acSaveNo
End Sub
