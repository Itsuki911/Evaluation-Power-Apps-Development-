Private Sub Form_Load()
    Me.cboFilterStatus.Value = "All"
    Me.txtFilterWorkDate.Value = ""

    If Not IsNull(Me.OpenArgs) And Len(Trim$(CStr(Me.OpenArgs))) > 0 Then
        Me.cboFilterEmployee.Value = Trim$(CStr(Me.OpenArgs))
        ApplyAttendanceFilter
    Else
        Me.cboFilterEmployee.Value = Null
        Me.Filter = ""
        Me.FilterOn = False
    End If
End Sub

Public Sub ApplyAttendanceFilter()
    Dim filterParts As String
    Dim empId As String
    Dim workDateText As String
    Dim statusVal As String
    Dim errMsg As String

    empId = Trim$(Nz(Me.cboFilterEmployee.Value, ""))
    workDateText = Trim$(Nz(Me.txtFilterWorkDate.Value, ""))
    statusVal = Trim$(Nz(Me.cboFilterStatus.Value, "All"))

    If Not ValidateDateFilter(workDateText, errMsg) Then
        MsgBox errMsg, vbExclamation, "検索条件エラー"
        Me.txtFilterWorkDate.SetFocus
        Exit Sub
    End If

    filterParts = ""

    If Len(empId) > 0 Then
        filterParts = AppendFilterCondition(filterParts, "EmployeeID = '" & SqlEscape(empId) & "'")
    End If

    If Len(workDateText) > 0 Then
        filterParts = AppendFilterCondition(filterParts, "WorkDate = '" & SqlEscape(workDateText) & "'")
    End If

    If Len(statusVal) > 0 And statusVal <> "All" Then
        filterParts = AppendFilterCondition(filterParts, "Status = '" & SqlEscape(statusVal) & "'")
    End If

    If Len(filterParts) > 0 Then
        Me.Filter = filterParts
        Me.FilterOn = True
    Else
        Me.Filter = ""
        Me.FilterOn = False
    End If
End Sub

Private Function AppendFilterCondition(ByVal existingFilter As String, ByVal newCondition As String) As String
    If Len(existingFilter) = 0 Then
        AppendFilterCondition = newCondition
    Else
        AppendFilterCondition = existingFilter & " AND " & newCondition
    End If
End Function

Private Sub cboFilterEmployee_AfterUpdate()
    ApplyAttendanceFilter
End Sub

Private Sub cboFilterStatus_AfterUpdate()
    ApplyAttendanceFilter
End Sub

Private Sub btnApplyFilter_Click()
    ApplyAttendanceFilter
End Sub

Private Sub btnClearFilter_Click()
    Me.cboFilterEmployee.Value = Null
    Me.txtFilterWorkDate.Value = ""
    Me.cboFilterStatus.Value = "All"
    Me.Filter = ""
    Me.FilterOn = False
End Sub

Private Sub btnPrintReport_Click()
    Dim whereCond As String
    If Me.FilterOn Then
        whereCond = Me.Filter
    Else
        whereCond = ""
    End If

    DoCmd.OpenReport "rptAttendanceHistory", acViewPreview, , whereCond
End Sub

Private Sub btnBackToMain_Click()
    If Not CurrentProject.AllForms("frmMain").IsLoaded Then
        DoCmd.OpenForm "frmMain", acNormal
    Else
        Forms("frmMain").RefreshAttendanceViews
        Forms("frmMain").SetFocus
    End If
    DoCmd.Close acForm, Me.Name, acSaveNo
End Sub
