' Legacy lock guard: prevent direct edits on closed months even if AllowEdits is toggled.
Private Sub Form_BeforeUpdate(Cancel As Integer)
    If Trim$(Nz(Me.txtCloseFlag.Value, CLOSE_OPEN)) = CLOSE_LOCKED Then
        MsgBox "この月の勤怠は既に月次締済のため直接変更できません。修正申請画面から申請してください。", vbExclamation, "締済ロック"
        Cancel = True
        Me.Undo
    End If
End Sub

' Launch frmCorrectionRequest passing pipe-delimited OpenArgs: "AttendanceID|EmployeeID|WorkDate"
Private Sub btnRequestFix_Click()
    Dim argStr As String
    argStr = Trim$(Nz(Me.txtAttendanceID.Value, "")) & "|" & _
             Trim$(Nz(Me.txtEmployeeID.Value, "")) & "|" & _
             Trim$(Nz(Me.txtWorkDate.Value, ""))
    DoCmd.OpenForm "frmCorrectionRequest", acNormal, , , , acWindowNormal, argStr
End Sub
