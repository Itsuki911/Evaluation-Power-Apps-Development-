Private Sub Form_Load()
    Me.cboFilterDepartment.Value = Null
    Me.Check31.Value = False
    ApplyEmployeeFilter
End Sub

Private Sub ApplyEmployeeFilter()
    Dim parts As String
    Dim dept As String

    parts = ""
    dept = Trim$(Nz(Me.cboFilterDepartment.Value, ""))
    If Len(dept) > 0 Then
        parts = "Department = '" & SqlEscape(dept) & "'"
    End If

    If Nz(Me.Check31.Value, False) = True Then
        If Len(parts) > 0 Then parts = parts & " AND "
        parts = parts & "Status = '" & EMP_STATUS_ACTIVE & "'"
    End If

    If Len(parts) > 0 Then
        Me.Filter = parts
        Me.FilterOn = True
    Else
        Me.Filter = ""
        Me.FilterOn = False
    End If
End Sub

Private Sub cboFilterDepartment_AfterUpdate()
    ApplyEmployeeFilter
End Sub

' Legacy control Check31 = Active employees only checkbox
Private Sub Check31_AfterUpdate()
    ApplyEmployeeFilter
End Sub

Private Sub btnClearFilter_Click()
    Me.cboFilterDepartment.Value = Null
    Me.Check31.Value = False
    ApplyEmployeeFilter
End Sub

Private Sub btnSelectForPunch_Click()
    Dim empId As String
    Dim errMsg As String

    empId = Nz(Me.txtEmployeeID.Value, "")
    If Not ValidateEmployeeActive(empId, errMsg) Then
        MsgBox errMsg, vbExclamation, "打刻対象外"
        Exit Sub
    End If

    If Not CurrentProject.AllForms("frmMain").IsLoaded Then
        DoCmd.OpenForm "frmMain", acNormal, , , , acWindowNormal, empId
    Else
        Forms("frmMain").SelectEmployee empId
        Forms("frmMain").SetFocus
    End If
End Sub

Private Sub btnShowHistory_Click()
    Dim empId As String
    Dim argStr As String

    empId = Nz(Me.txtEmployeeID.Value, "")
    If Len(Trim$(empId)) = 0 Then Exit Sub

    argStr = empId & "|" & Format$(Date, "yyyy-mm") & "|DETAIL"
    DoCmd.OpenForm "frmAttendance", acNormal, , , , acWindowNormal, argStr
End Sub

Private Sub btnBackToMain_Click()
    If Not CurrentProject.AllForms("frmMain").IsLoaded Then
        DoCmd.OpenForm "frmMain", acNormal
    Else
        Forms("frmMain").SetFocus
    End If
    DoCmd.Close acForm, Me.Name, acSaveNo
End Sub
