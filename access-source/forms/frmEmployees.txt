Attribute VB_GlobalNameSpace = False

Option Compare Database

Private Sub Form_Load()
    Me.Filter = ""
    Me.FilterOn = False
    Me.cboFilterDepartment.Value = Null
End Sub

Private Sub cboFilterDepartment_AfterUpdate()
    Dim selectedDept As String

    selectedDept = Trim$(Nz(Me.cboFilterDepartment.Value, ""))
    If Len(selectedDept) = 0 Then
        Me.Filter = ""
        Me.FilterOn = False
    Else
        Me.Filter = "Department = '" & SqlEscape(selectedDept) & "'"
        Me.FilterOn = True
    End If
End Sub

Private Sub btnClearFilter_Click()
    Me.cboFilterDepartment.Value = Null
    Me.Filter = ""
    Me.FilterOn = False
End Sub

Private Sub btnSelectForPunch_Click()
    Dim empId As String
    Dim errMsg As String

    empId = Nz(Me.txtEmployeeID.Value, "")
    If Not ValidateEmployeeActive(empId, errMsg) Then
        MsgBox errMsg, vbExclamation, "打刻対象外"
        Exit Sub
    End If

    If Not CurrentProject.AllForms("frmMain").IsLoaded Then
        DoCmd.OpenForm "frmMain", acNormal
    End If

    Forms("frmMain").SelectEmployee empId
    Forms("frmMain").SetFocus
End Sub

Private Sub btnShowHistory_Click()
    Dim empId As String

    empId = Nz(Me.txtEmployeeID.Value, "")
    If Len(Trim$(empId)) = 0 Then
        Exit Sub
    End If

    DoCmd.OpenForm "frmAttendance", acNormal, , , , acWindowNormal, empId
End Sub

Private Sub btnBackToMain_Click()
    If Not CurrentProject.AllForms("frmMain").IsLoaded Then
        DoCmd.OpenForm "frmMain", acNormal
    Else
        Forms("frmMain").SetFocus
    End If
    DoCmd.Close acForm, Me.Name, acSaveNo
End Sub
