Attribute VB_GlobalNameSpace = False

Attribute VB_PredeclaredId = True
Attribute VB_Exposed = False
Option Compare Database
Option Explicit

Private Sub Form_Load()
    Dim activeUser As String

    activeUser = GetActiveUserID()
    If Len(activeUser) = 0 Then
        InitAppContext
        activeUser = GetActiveUserID()
    End If

    Me.cboOperator.Value = activeUser
    SyncRoleDisplay
End Sub

Private Sub SyncRoleDisplay()
    Dim empId As String
    Dim roleText As String
    Dim canAdmin As Boolean

    empId = Trim$(Nz(Me.cboOperator.Value, ""))
    SetSessionUser empId
    roleText = GetActiveUserRole()
    Me.txtHiddenRole.Value = roleText
    Me.lblRoleBadge.Caption = "権限: " & roleText

    canAdmin = IsManagerOrAdmin(empId)
    Me.btnMenuMonthlyClose.Enabled = canAdmin
    Me.Recalc
End Sub

Private Sub cboOperator_AfterUpdate()
    SyncRoleDisplay
End Sub

Private Sub btnMenuMonthlyClose_Click()
    Dim empId As String
    empId = GetActiveUserID()
    If Not IsManagerOrAdmin(empId) Then
        MsgBox "月次締め画面は Manager または Admin 権限のみ利用できます。", vbExclamation, "権限エラー"
        Exit Sub
    End If
    DoCmd.OpenForm "frmMonthlyClose", acNormal
End Sub

Private Sub btnExitApp_Click()
    If MsgBox("勤怠管理システムを終了しますか？", vbYesNo + vbQuestion, "終了確認") = vbYes Then
        DoCmd.Quit acQuitSaveNone
    End If
End Sub
