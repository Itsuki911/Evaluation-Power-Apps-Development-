Attribute VB_GlobalNameSpace = False

Private Sub Form_Load()
    Me.lblTodayDate.Caption = "本日: " & Format$(Date, "yyyy-mm-dd")
    Me.txtSelectedEmployeeName.Value = ""
    Me.txtSelectedDepartment.Value = ""
    Me.txtCurrentStatus.Value = "-"
    RefreshAttendanceViews
End Sub

Public Sub SelectEmployee(ByVal employeeId As String)
    Me.cboEmployee.Requery
    Me.cboEmployee.Value = employeeId
    UpdateSelectedEmployeeDisplay
    RefreshSelectedEmployeeHistory employeeId
End Sub

Public Sub RefreshAttendanceViews()
    Dim selectedId As String

    Me.cboEmployee.Requery
    Me.lstOpenAttendance.Requery

    selectedId = Nz(Me.cboEmployee.Value, "")
    UpdateSelectedEmployeeDisplay
    RefreshSelectedEmployeeHistory selectedId
End Sub

Private Sub UpdateSelectedEmployeeDisplay()
    Dim selectedId As String

    selectedId = Trim$(Nz(Me.cboEmployee.Value, ""))
    If Len(selectedId) = 0 Then
        Me.txtSelectedEmployeeName.Value = ""
        Me.txtSelectedDepartment.Value = ""
        Me.txtCurrentStatus.Value = "-"
        Exit Sub
    End If

    Me.txtSelectedEmployeeName.Value = Nz(DLookup("EmployeeName", "Employees", "EmployeeID = '" & SqlEscape(selectedId) & "'"), "")
    Me.txtSelectedDepartment.Value = Nz(DLookup("Department", "Employees", "EmployeeID = '" & SqlEscape(selectedId) & "'"), "")
    Me.txtCurrentStatus.Value = GetEmployeeCurrentStatus(selectedId)
End Sub

Private Sub RefreshSelectedEmployeeHistory(ByVal employeeId As String)
    Dim safeId As String

    safeId = SqlEscape(Trim$(employeeId))
    If Len(safeId) = 0 Then
        Me.lstEmployeeHistory.RowSource = _
            "SELECT AttendanceID, WorkDate, ClockIn, ClockOut, Status " & _
            "FROM qryAttendanceHistory WHERE 1 = 0;"
    Else
        Me.lstEmployeeHistory.RowSource = _
            "SELECT AttendanceID, WorkDate, ClockIn, ClockOut, Status " & _
            "FROM qryAttendanceHistory " & _
            "WHERE EmployeeID = '" & safeId & "' " & _
            "ORDER BY WorkDate DESC, ClockIn DESC, AttendanceID DESC;"
    End If
    Me.lstEmployeeHistory.Requery
End Sub

Private Sub cboEmployee_AfterUpdate()
    Dim selectedId As String
    selectedId = Nz(Me.cboEmployee.Value, "")
    UpdateSelectedEmployeeDisplay
    RefreshSelectedEmployeeHistory selectedId
    Me.lblStatusMessage.Caption = "選択従業員: " & selectedId & " (" & Nz(Me.txtSelectedEmployeeName.Value, "") & ")"
End Sub

Private Sub btnClockIn_Click()
    Dim selectedId As String
    Dim resultMsg As String
    Dim createdId As String
    Dim isSuccess As Boolean

    selectedId = Nz(Me.cboEmployee.Value, "")
    isSuccess = ClockInEmployee(selectedId, resultMsg, createdId)

    If isSuccess Then
        RefreshAttendanceViews
        Me.lblStatusMessage.Caption = "出勤打刻完了: " & selectedId & " (" & createdId & ")"
        MsgBox resultMsg, vbInformation, "出勤打刻"
    Else
        Me.lblStatusMessage.Caption = "出勤打刻不可: " & resultMsg
        MsgBox resultMsg, vbExclamation, "出勤打刻エラー"
    End If
End Sub

Private Sub btnClockOut_Click()
    Dim selectedId As String
    Dim resultMsg As String
    Dim updatedId As String
    Dim isSuccess As Boolean

    selectedId = Nz(Me.cboEmployee.Value, "")
    isSuccess = ClockOutEmployee(selectedId, resultMsg, updatedId)

    If isSuccess Then
        RefreshAttendanceViews
        Me.lblStatusMessage.Caption = "退勤打刻完了: " & selectedId & " (" & updatedId & ")"
        MsgBox resultMsg, vbInformation, "退勤打刻"
    Else
        Me.lblStatusMessage.Caption = "退勤打刻不可: " & resultMsg
        MsgBox resultMsg, vbExclamation, "退勤打刻エラー"
    End If
End Sub

Private Sub btnRefresh_Click()
    RefreshAttendanceViews
    Me.lblStatusMessage.Caption = "最新データを再読み込みしました (" & Format$(Now(), "yyyy-mm-dd hh:nn:ss") & ")。"
End Sub

Private Sub lstOpenAttendance_DblClick(Cancel As Integer)
    Dim selectedId As String
    selectedId = Nz(Me.lstOpenAttendance.Value, "")
    If Len(selectedId) > 0 Then
        SelectEmployee selectedId
        Me.lblStatusMessage.Caption = "未退勤一覧から従業員 " & selectedId & " を選択しました。"
    End If
End Sub

Private Sub btnOpenEmployees_Click()
    DoCmd.OpenForm "frmEmployees", acNormal
End Sub

Private Sub btnOpenAttendance_Click()
    DoCmd.OpenForm "frmAttendance", acNormal
End Sub

Private Sub btnViewFullHistory_Click()
    Dim selectedId As String
    Dim errMsg As String

    selectedId = Nz(Me.cboEmployee.Value, "")
    If Not ValidateEmployeeSelected(selectedId, errMsg) Then
        MsgBox errMsg, vbExclamation, "従業員未選択"
        Exit Sub
    End If

    DoCmd.OpenForm "frmAttendance", acNormal, , , , acWindowNormal, selectedId
End Sub

Private Sub btnOpenReport_Click()
    DoCmd.OpenReport "rptAttendanceHistory", acViewPreview
End Sub
