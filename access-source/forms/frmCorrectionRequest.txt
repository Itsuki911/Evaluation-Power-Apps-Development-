Private Sub Form_Load()
    Dim argAttId As String
    Dim argEmpId As String
    Dim argDate As String
    Dim activeUser As String
    Dim canApprove As Boolean

    activeUser = GetActiveUserID()
    canApprove = IsManagerOrAdmin(activeUser)

    Me.lblCurrentOperator.Caption = "操作者: " & activeUser & " (" & GetActiveUserRole() & ")"
    Me.btnApprove.Enabled = canApprove
    Me.btnReject.Enabled = canApprove

    ' Parse pipe-delimited OpenArgs: "AttendanceID|EmployeeID|TargetDate"
    argAttId = ParseOpenArgToken(Me.OpenArgs, 0)
    argEmpId = ParseOpenArgToken(Me.OpenArgs, 1)
    argDate = ParseOpenArgToken(Me.OpenArgs, 2)

    Me.txtAttendanceID.Value = argAttId
    If Len(argEmpId) > 0 Then
        Me.cboReqEmployee.Value = argEmpId
    Else
        Me.cboReqEmployee.Value = activeUser
    End If

    If Len(argDate) > 0 Then
        Me.txtTargetDate.Value = argDate
    Else
        Me.txtTargetDate.Value = Format$(Date, "yyyy-mm-dd")
    End If

    Me.cboReqWorkType.Value = WT_NORMAL
    Me.txtReqBreakMin.Value = "60"

    ' Pre-fill existing timestamps when AttendanceID is supplied
    If Len(argAttId) > 0 Then
        Me.txtReqClockIn.Value = Nz(DLookup("ClockIn", "AttendanceRecords", "AttendanceID = '" & SqlEscape(argAttId) & "'"), "")
        Me.txtReqClockOut.Value = Nz(DLookup("ClockOut", "AttendanceRecords", "AttendanceID = '" & SqlEscape(argAttId) & "'"), "")
    End If

    Me.lstPendingRequests.Requery
End Sub

Private Sub btnSubmitRequest_Click()
    Dim msg As String
    Dim newReqId As String
    Dim ok As Boolean

    ok = SubmitCorrectionRequest( _
        Nz(Me.txtAttendanceID.Value, ""), _
        Nz(Me.cboReqEmployee.Value, ""), _
        Nz(Me.txtTargetDate.Value, ""), _
        Nz(Me.txtReqClockIn.Value, ""), _
        Nz(Me.txtReqClockOut.Value, ""), _
        Nz(Me.txtReqBreakMin.Value, ""), _
        Nz(Me.cboReqWorkType.Value, WT_NORMAL), _
        Nz(Me.txtReason.Value, ""), _
        msg, _
        newReqId)

    If ok Then
        Me.txtReason.Value = ""
        Me.lstPendingRequests.Requery
        Me.lblResultInfo.Caption = msg
        MsgBox msg, vbInformation, "修正申請完了"
    Else
        Me.lblResultInfo.Caption = msg
        MsgBox msg, vbExclamation, "入力エラー"
    End If
End Sub

Private Sub btnApprove_Click()
    ProcessSelectedRequest APPR_APPROVED
End Sub

Private Sub btnReject_Click()
    ProcessSelectedRequest APPR_REJECTED
End Sub

Private Sub ProcessSelectedRequest(ByVal targetStatus As String)
    Dim reqId As String
    Dim approverId As String
    Dim msg As String

    reqId = Trim$(Nz(Me.lstPendingRequests.Value, ""))
    If Len(reqId) = 0 Then
        MsgBox "処理対象の未承認申請を一覧から選択してください。", vbExclamation, "未選択"
        Exit Sub
    End If

    approverId = GetActiveUserID()
    If ApproveOrRejectCorrection(reqId, approverId, targetStatus, msg) Then
        Me.lstPendingRequests.Requery
        Me.lblResultInfo.Caption = msg
        MsgBox msg, vbInformation, "承認処理完了"
    Else
        Me.lblResultInfo.Caption = msg
        MsgBox msg, vbExclamation, "承認処理エラー"
    End If
End Sub

Private Sub btnReloadPending_Click()
    Me.lstPendingRequests.Requery
    Me.lblResultInfo.Caption = "未承認申請一覧を再読込しました。"
End Sub

Private Sub btnCloseForm_Click()
    DoCmd.Close acForm, Me.Name, acSaveNo
End Sub
