Private Sub Form_Load()
    Me.txtCloseYM.Value = Format$(Date, "yyyy-mm")
    Me.txtCloseNote.Value = ""
    Me.Recalc
End Sub

Private Sub txtCloseYM_AfterUpdate()
    Me.Recalc
End Sub

Private Sub btnExecuteClose_Click()
    Dim ymText As String
    Dim noteText As String
    Dim msg As String

    ymText = Trim$(Nz(Me.txtCloseYM.Value, ""))
    noteText = Trim$(Nz(Me.txtCloseNote.Value, ""))

    If ExecuteMonthlyClose(ymText, GetActiveUserID(), CLOSE_LOCKED, noteText, msg) Then
        Me.Requery
        Me.Recalc
        Me.lblCloseStatusMsg.Caption = msg
        MsgBox msg, vbInformation, "月次締め確定"
    Else
        Me.lblCloseStatusMsg.Caption = msg
        MsgBox msg, vbExclamation, "月次締め不可"
    End If
End Sub

Private Sub btnReopenMonth_Click()
    Dim ymText As String
    Dim noteText As String
    Dim msg As String

    ymText = Trim$(Nz(Me.txtCloseYM.Value, ""))
    noteText = Trim$(Nz(Me.txtCloseNote.Value, ""))

    If MsgBox(ymText & " の月次締めを解除しますか？", vbYesNo + vbQuestion, "締め解除確認") <> vbYes Then
        Exit Sub
    End If

    If ExecuteMonthlyClose(ymText, GetActiveUserID(), CLOSE_OPEN, noteText, msg) Then
        Me.Requery
        Me.Recalc
        Me.lblCloseStatusMsg.Caption = msg
        MsgBox msg, vbInformation, "締め解除完了"
    Else
        Me.lblCloseStatusMsg.Caption = msg
        MsgBox msg, vbExclamation, "締め解除エラー"
    End If
End Sub

Private Sub btnExportPayrollCsv_Click()
    Dim ymText As String
    Dim outPath As String
    Dim msg As String

    ymText = Trim$(Nz(Me.txtCloseYM.Value, ""))
    outPath = CurrentProject.Path & "\payroll_" & Replace(ymText, "-", "") & ".csv"

    If ExportMonthlyPayrollCsv(ymText, outPath, msg) Then
        Me.lblCloseStatusMsg.Caption = msg
        MsgBox msg, vbInformation, "給与CSV出力"
    Else
        Me.lblCloseStatusMsg.Caption = msg
        MsgBox msg, vbExclamation, "CSV出力エラー"
    End If
End Sub

Private Sub btnCloseForm_Click()
    DoCmd.Close acForm, Me.Name, acSaveNo
End Sub
